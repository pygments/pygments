component Counter {
  state counter = 0

  fun increment {
    next { counter: counter + 1 }
  }

  fun render {
    <div>
      <button onClick={decrement}>"Decrement"</button>
      <span>"#{counter}"</span>
    </div>
  }

  style root {
    color: red;
  }
}
