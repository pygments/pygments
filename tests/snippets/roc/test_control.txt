---input---
# a comment
## a doc comment
simple_match = |color| {
    match color {
        Red => "The color is red."
        Green => "The color is green."
        Blue | BabyBlue => "The color is blue."
        [1, 2, ..] => 66
        [1, .. as tail] => 77 + tail.len()
        _ => 100
    }
}
count_down = |limit| {
    var $count = 0
    while $count < limit {
        $count = $count + 1
        if $count == 3 {
            break
        } else {
            {}
        }
    }
    $count
}
for_loop = |num_list| {
    var $sum = 0
    for num in num_list {
        $sum = $sum + num
    }
    $sum
}
not_implemented = |str| {
    if str == "" {
        crash "not implemented"
    } else {
        return str
    }
}
check = |flag| {
    expect flag or (!flag)
    dbg flag
    flag and True
}
first_or = |strings| {
    first = strings.first() ? NoFirstError
    first?
}
placeholder = |_x| ...
